0) რატომ ვიყენებთ callback ფუნქციას:
// Callback არის ფუნქცია, რომელსაც სხვა ფუნქციას არგუმენტად გადავცემთ.
// მას ვიყენებთ, რათა განვსაზღვროთ რა მოქმედება შესრულდეს ძირითადი ფუნქციის დასრულების შემდეგ.
// ეს განსაკუთრებით სასარგებლოა ასინქრონული ოპერაციებისას (მაგალითად, მონაცემების ჩატვირთვისას),
// რადგან შედეგის მიღების შემდეგ callback-ს ვიძახებთ და საჭირო მოქმედებას ვასრულებთ.

function greet(name, callback) {
	console.log("გამარჯობა, " + name);
	callback(); // ვიძახებთ არგუმენტად მიღებულ ფუნქციას
}

greet("ანა", function () {
	console.log("მისალმება დასრულდა.");
});
